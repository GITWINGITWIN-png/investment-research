# finbg: automatic schedule (install with: crontab scripts/crontab.txt)
#
# Daily and weekly jobs go through anacron, which runs jobs missed while the machine was off (see scripts/anacrontab)
#   called every hour from 05:05 to 23:05 and at boot; not between 00:00 and 04:59, while the US market is open
#   anacron remembers whether today's run happened, so frequent calls never run a job twice
5 5-23 * * * /usr/sbin/anacron -s -t /home/you/investment-research/scripts/anacrontab -S /home/you/.local/state/finbg/anacron
@reboot sleep 120; [ "$(date +\%H)" -ge 5 ] && /usr/sbin/anacron -s -t /home/you/investment-research/scripts/anacrontab -S /home/you/.local/state/finbg/anacron
